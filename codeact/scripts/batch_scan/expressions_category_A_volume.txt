# A2: volume ratio patterns
rank(volume/ts_mean(volume, 20))

rank(ts_sum(volume, 5)/ts_sum(volume, 20))
rank(ts_sum(volume, 5)/ts_sum(volume, 60))
# A3: volume delta patterns
rank(ts_delta(ts_mean(volume, 5), 3))
rank(ts_delta(ts_mean(volume, 20), 5))
rank(ts_delta(ts_mean(volume, 60), 10))
# A4: volume rank delta patterns
rank(ts_delta(ts_rank(volume, 5), 3))
rank(ts_delta(ts_rank(volume, 20), 5))
rank(ts_delta(ts_rank(volume, 60), 10))
# A5: volume std/volatility
